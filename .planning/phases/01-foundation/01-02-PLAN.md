**Type:** create
**File:** `app/src/main/java/com/example/player/data/local/entity/Track.kt`

```kotlin
package com.example.player.data.local.entity

import androidx.room.Entity
import androidx.room.PrimaryKey
import java.util.UUID

@Entity(tableName = "tracks")
data class Track(
    @PrimaryKey
    val id: String = UUID.randomUUID().toString(),
    val uri: String,                    // content URI o path del file
    val title: String,
    val artist: String,
    val album: String,
    val duration: Long,                 // millisecondi
    val year: String?,
    val trackNumber: String?,
    val albumArtPath: String?,          // path locale alla copertina salvata
    val importedAt: Long = System.currentTimeMillis()
)
```

**Done:** File creato con tutti i campi.

### Task 2: DAO `TrackDao`
**Type:** create
**File:** `app/src/main/java/com/example/player/data/local/dao/TrackDao.kt`

```kotlin
package com.example.player.data.local.dao

import androidx.room.*
import com.example.player.data.local.entity.Track
import kotlinx.coroutines.flow.Flow

@Dao
interface TrackDao {
    @Query("SELECT * FROM tracks ORDER BY title ASC")
    fun getAllTracks(): Flow<List<Track>>

    @Query("SELECT * FROM tracks WHERE id = :id")
    suspend fun getTrackById(id: String): Track?

    @Insert(onConflict = OnConflictStrategy.REPLACE)
    suspend fun insertTrack(track: Track)

    @Insert(onConflict = OnConflictStrategy.REPLACE)
    suspend fun insertTracks(tracks: List<Track>)

    @Delete
    suspend fun deleteTrack(track: Track)

    @Query("SELECT * FROM tracks WHERE uri = :uri LIMIT 1")
    suspend fun getTrackByUri(uri: String): Track?
}
```

**Done:** DAO con Flow per lista reattiva, query per URI (evita duplicati).

### Task 3: `PlayerDatabase`
**Type:** create
**File:** `app/src/main/java/com/example/player/data/local/PlayerDatabase.kt`

```kotlin
package com.example.player.data.local

import androidx.room.Database
import androidx.room.RoomDatabase
import com.example.player.data.local.dao.TrackDao
import com.example.player.data.local.entity.Track

@Database(
    entities = [Track::class],
    version = 1,
    exportSchema = false
)
abstract class PlayerDatabase : RoomDatabase() {
    abstract fun trackDao(): TrackDao

    companion object {
        const val DATABASE_NAME = "player_db"
    }
}
```

**Done:** Database definito, pronto per essere iniettato.
